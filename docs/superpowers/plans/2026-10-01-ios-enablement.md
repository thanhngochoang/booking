**Goal:** The Flutter app builds and runs on the iOS Simulator and on an iPhone, with Firebase, email, Google and Facebook sign-in working, CI building iOS on every push, and an iOS section in the battery and performance guide.

**Architecture:** iOS configuration that holds secrets or personal values (Google client ids, Facebook token, Apple team) comes from a gitignored `ios/Flutter/Secrets.xcconfig` read by `Info.plist` through `$(VARIABLE)` references; a committed `Secrets.xcconfig.example` lists every variable, and a Dart test keeps the two in sync. Firebase options for iOS are generated by `flutterfire configure` into the already-gitignored `lib/firebase_options.dart`. A platform-neutral `integration_test` smoke test runs on the Simulator (and on Android) with fake repositories. CI gains a macOS job that builds for the Simulator without signing.

**Tech Stack:** Xcode (latest stable from the App Store), iOS 15.0 minimum (already in `project.pbxproj`), CocoaPods through Bundler (project-local) or Swift Package Manager, FlutterFire CLI, `integration_test`, GitHub Actions `macos-latest`.

**Spec:** `docs/superpowers/specs/2026-09-30-photography-marketplace-design.md` (platforms: Android 23+, iOS 13+ in the original; the project file already says iOS 15.0, which this plan keeps), `CLAUDE.md` (Flutter rewrite rules), `docs/testing/battery-and-performance.md` (created by the screen-codes plan).

**Prerequisite:** none for Tasks 1–4 (they run on any machine). Tasks 5–8 need a Mac with Xcode and the person's own accounts; they are marked **[người dùng]** where only the owner can do the step.

```dart
// test/platform/ios_config_test.dart
import 'dart:io';

import 'package:flutter_test/flutter_test.dart';

final _plist = File('ios/Runner/Info.plist');
final _example = File('ios/Flutter/Secrets.xcconfig.example');

/// `<key>K</key>` followed by its `<string>` value.
String? _stringFor(String plist, String key) => RegExp(
  '<key>${RegExp.escape(key)}</key>\\s*<string>([^<]*)</string>',
).firstMatch(plist)?.group(1);

void main() {
  late String plist;
  setUpAll(() => plist = _plist.readAsStringSync());

  test('display name is Vietnamese with diacritics', () {
    expect(_stringFor(plist, 'CFBundleDisplayName'), 'Nhiếp ảnh gia');
  });

  test('Google and Facebook sign-in keys read from xcconfig variables', () {
    expect(_stringFor(plist, 'GIDClientID'), r'$(GOOGLE_IOS_CLIENT_ID)');
    expect(_stringFor(plist, 'FacebookAppID'), r'$(FACEBOOK_APP_ID)');
    expect(_stringFor(plist, 'FacebookClientToken'), r'$(FACEBOOK_CLIENT_TOKEN)');
    expect(_stringFor(plist, 'FacebookDisplayName'), isNotEmpty);
    final urlTypes = RegExp(r'<key>CFBundleURLTypes</key>\s*<array>([\s\S]*?)</array>\s*<key>')
        .firstMatch(plist)
        ?.group(1);
    expect(urlTypes, contains(r'<string>$(GOOGLE_REVERSED_CLIENT_ID)</string>'));
    expect(urlTypes, contains(r'<string>fb$(FACEBOOK_APP_ID)</string>'));
  });

  test('Facebook login can see the Facebook app', () {
    final queries = RegExp(r'<key>LSApplicationQueriesSchemes</key>\s*<array>([\s\S]*?)</array>')
        .firstMatch(plist)
        ?.group(1);
    for (final s in ['fbapi', 'fb-messenger-share-api', 'fbauth2', 'fbshareextension']) {
      expect(queries, contains('<string>$s</string>'), reason: s);
    }
  });

  test('no secret is written literally into Info.plist', () {
    // Facebook app ids are 15-16 digits; Google client ids end in this suffix.
    expect(RegExp(r'\b\d{15,16}\b').hasMatch(plist), isFalse);
    expect(plist.contains('.apps.googleusercontent.com'), isFalse);
  });

  test('every $(VAR) used by Info.plist is listed in Secrets.xcconfig.example', () {
    final used = RegExp(r'\$\(([A-Z_]+)\)')
        .allMatches(plist)
        .map((m) => m.group(1)!)
        .where((v) => v.startsWith('GOOGLE_') || v.startsWith('FACEBOOK_'))
        .toSet();
    final example = _example.readAsStringSync();
    for (final v in used) {
      expect(RegExp('^$v\\s*=', multiLine: true).hasMatch(example), isTrue, reason: v);
    }
    expect(RegExp(r'^DEVELOPMENT_TEAM\s*=', multiLine: true).hasMatch(example), isTrue);
  });

  test('both build configurations include the optional secrets file', () {
    for (final f in ['ios/Flutter/Debug.xcconfig', 'ios/Flutter/Release.xcconfig']) {
      expect(File(f).readAsStringSync(), contains('#include? "Secrets.xcconfig"'), reason: f);
    }
  });

  test('secrets and Firebase files are gitignored', () {
    final ignore = File('.gitignore').readAsStringSync();
    for (final p in [
      '/ios/Flutter/Secrets.xcconfig',
      '/ios/Runner/GoogleService-Info.plist',
      '/lib/firebase_options.dart',
    ]) {
      expect(ignore, contains(p), reason: p);
    }
  });

  test('no background location or background modes', () {
    expect(plist.contains('NSLocationAlwaysAndWhenInUseUsageDescription'), isFalse);
    expect(plist.contains('UIBackgroundModes'), isFalse);
  });
}
```

```dart
// integration_test/app_smoke_test.dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:shared_preferences/shared_preferences.dart';

import 'package:photobooking/app/app.dart';
import 'package:photobooking/app/router.dart';
import 'package:photobooking/core/core.dart';
import 'package:photobooking/data/auth/auth_providers.dart';
import 'package:photobooking/data/auth/auth_repository.dart';
import 'package:photobooking/data/user/user_repository.dart';
import 'package:photobooking/features/settings/theme_mode_controller.dart';

/// Runs the real widget tree on a device with fake repositories, so it needs
/// no Firebase project. It proves the app starts, lays out and navigates on
/// the platform (fonts, safe areas, text input, back gestures).
void main() {
  IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  testWidgets('register, pick a role, reach the tabs and settings', (tester) async {
    SharedPreferences.setMockInitialValues({});
    final prefs = await SharedPreferences.getInstance();
    await tester.pumpWidget(
      ProviderScope(
        overrides: [
          authRepositoryProvider.overrideWithValue(FakeAuthRepository()),
          userRepositoryProvider.overrideWithValue(FakeUserRepository()),
          sharedPreferencesProvider.overrideWithValue(prefs),
        ],
        child: Consumer(
          builder: (context, ref, _) => MyApp(
            router: ref.watch(routerProvider),
            theme: buildLightTheme(),
            darkTheme: buildDarkTheme(),
          ),
        ),
      ),
    );
    await tester.pumpAndSettle();

    // Login screen (S28) → register (S41).
    await tester.tap(find.byKey(const Key('login-register')));
    await tester.pumpAndSettle();
    await tester.enterText(find.byKey(const Key('register-name')), 'Minh Thư');
    await tester.enterText(find.byKey(const Key('register-email')), 'minh@example.test');
    await tester.enterText(find.byKey(const Key('register-password')), 'password1');
    await tester.enterText(find.byKey(const Key('register-confirm')), 'password1');
    await tester.tap(find.byKey(const Key('register-submit')));
    await tester.pumpAndSettle();

    // Role (S29) → customer → tabs.
    await tester.tap(find.byKey(const Key('role-customer')));
    await tester.pumpAndSettle();
    expect(find.byType(NavigationBar), findsOneWidget);

    // Profile tab → Settings (S31) → change theme.
    await tester.tap(find.text('Hồ sơ'));
    await tester.pumpAndSettle();
    await tester.tap(find.byIcon(Icons.settings_outlined));
    await tester.pumpAndSettle();
    await tester.tap(find.byKey(const Key('theme-light')));
    await tester.pumpAndSettle();
    expect(prefs.getString('themeMode'), 'light');
  });
}
```

- [ ] **Step 2: Write the doctor script**

```bash
#!/usr/bin/env bash
# scripts/ios-doctor.sh — is this Mac ready to build the iOS app?
set -u
cd "$(dirname "$0")/../app_flutter" || exit 1
ok=0
check() { if eval "$2" >/dev/null 2>&1; then echo "✓ $1"; else echo "✗ $1 — $3"; ok=1; fi; }

check "Xcode selected" "xcodebuild -version" \
  "install Xcode, then: sudo xcode-select --switch /Applications/Xcode.app/Contents/Developer"
check "iOS Simulator runtime" "xcrun simctl list runtimes | grep -q 'iOS'" \
  "xcodebuild -downloadPlatform iOS"
check "CocoaPods (Bundler) or Swift Package Manager" \
  "(cd ios && bundle exec pod --version) || ../.flutter/bin/flutter config --list | grep -q 'enable-swift-package-manager: true'" \
  "cd app_flutter/ios && bundle install (see Task 5 Step 3)"
check "ios/Flutter/Secrets.xcconfig" "test -f ios/Flutter/Secrets.xcconfig" \
  "cp ios/Flutter/Secrets.xcconfig.example ios/Flutter/Secrets.xcconfig and fill it in"
check "GoogleService-Info.plist" "test -f ios/Runner/GoogleService-Info.plist" \
  "run flutterfire configure (Task 6)"
check "firebase_options.dart has iOS" "grep -q 'ios' lib/firebase_options.dart" \
  "run flutterfire configure (Task 6)"
exit $ok
```
